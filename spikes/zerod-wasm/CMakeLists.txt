cmake_minimum_required(VERSION 3.18)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
project(corosim_zerod LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(SV ${CMAKE_CURRENT_SOURCE_DIR}/vendor/svZeroDSolver/src)
include(FetchContent)
set(EIGEN_BUILD_DOC OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(Eigen GIT_REPOSITORY https://gitlab.com/libeigen/eigen.git GIT_TAG 3.4.0 GIT_SHALLOW TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.2/json.tar.xz)
FetchContent_MakeAvailable(Eigen json)
file(GLOB SRCS ${SV}/algebra/*.cpp ${SV}/model/*.cpp)
list(APPEND SRCS ${SV}/solve/SimulationParameters.cpp)
set(INC ${SV}/algebra ${SV}/model ${SV}/solve)
if(EMSCRIPTEN)
  add_executable(svzerod_wasm shim.cpp ${SRCS})
  target_include_directories(svzerod_wasm PRIVATE ${INC})
  target_link_libraries(svzerod_wasm PRIVATE Eigen3::Eigen nlohmann_json::nlohmann_json embind)
  target_compile_options(svzerod_wasm PRIVATE -O3 -fexceptions)
  target_link_options(svzerod_wasm PRIVATE -O3 -fexceptions
    -sMODULARIZE=1 -sEXPORT_ES6=0 -sEXPORT_NAME=createSvZeroD
    -sALLOW_MEMORY_GROWTH=1 -sEXPORTED_RUNTIME_METHODS=getExceptionMessage,decrementExceptionRefcount -sENVIRONMENT=web,worker,node -sASSERTIONS=0)
  set_target_properties(svzerod_wasm PROPERTIES SUFFIX ".js")
else()
  add_executable(shim_native shim.cpp ${SRCS})
  target_include_directories(shim_native PRIVATE ${INC})
  target_link_libraries(shim_native PRIVATE Eigen3::Eigen nlohmann_json::nlohmann_json)
  target_compile_options(shim_native PRIVATE -O3)
endif()
